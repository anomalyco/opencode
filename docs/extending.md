# Extending OpenCode

OpenCode is designed to be highly extensible, allowing developers to add custom tools, agents, plugins, and integrations to enhance functionality.

## Extension Points Overview

## Custom Tools

### Tool Definition

```typescript
/**
 * Defines the contract for all OpenCode tools
 * Provides type safety, validation, and execution context management
 *
 * @template Parameters - Zod schema type for tool input parameters
 * @template Metadata - Type for tool-specific execution metadata
 */
export interface Info<Parameters extends z.ZodType = z.ZodType, M extends Metadata = Metadata> = {
  /** Unique identifier for the tool (used in tool calls) */
  id: string

  /** Async initialization function that returns tool configuration */
  init: () => Promise<{
    /** Human-readable description of what the tool does */
    description: string

    /** Zod schema for validating tool input parameters */
    parameters: Parameters

    /** Core execution function with context and error handling */
    execute(
      args: z.infer<Parameters>,
      ctx: Context,
    ): Promise<{
      /** Human-readable title for the tool execution */
      title: string

      /** Tool-specific metadata for tracking and analytics */
      metadata: M

      /** Primary text output from the tool */
      output: string

      /** Optional file attachments generated by the tool */
      attachments?: MessageV2.FilePart[]
    }>

    /** Optional custom error formatter for validation failures */
    formatValidationError?(error: z.ZodError): string
  }>
}
```

### Tool Creation Helper

```typescript
/**
 * Helper function to create well-defined tools with automatic validation
 * Wraps user-provided logic with parameter validation and error handling
 *
 * @param id - Unique tool identifier
 * @param init - Tool configuration or async factory function
 * @returns Complete tool definition ready for registration
 */
export function define<Parameters extends z.ZodType, Result extends Metadata>(
  id: string,
  init: Info<Parameters, Result>["init"] | Awaited<ReturnType<Info<Parameters, Result>["init"]>>,
): Info<Parameters, Result> {
  return {
    id,
    init: async () => {
      // Support both sync and async initialization
      const toolInfo = init instanceof Function ? await init() : init
      const execute = toolInfo.execute

      // Wrap execution with automatic parameter validation
      toolInfo.execute = (args, ctx) => {
        try {
          // Validate all parameters against Zod schema
          toolInfo.parameters.parse(args)
        } catch (error) {
          // Use custom error formatter if provided, otherwise use default
          if (error instanceof z.ZodError && toolInfo.formatValidationError) {
            throw new Error(toolInfo.formatValidationError(error), { cause: error })
          }

          // Provide helpful error message with schema requirements
          throw new Error(
            `The ${id} tool was called with invalid arguments: ${error}.\n` +
              `Please rewrite the input so it satisfies the expected schema.`,
            { cause: error },
          )
        }

        // Execute the original tool logic with validated parameters
        return execute(args, ctx)
      }
      return toolInfo
    },
  }
}
```

### File Operations Tool Example

```typescript
/**
 * File processing tool with validation and security checks
 * Demonstrates best practices for tool development
 */
export const FileProcessorTool = define("file-processor", {
  description: "Process files with custom logic and validation",
  parameters: z.object({
    filePath: z.string().describe("Path to file to process"),
    operation: z.enum(["analyze", "transform", "validate"]).describe("Operation to perform on the file"),
    options: z.record(z.string(), z.any()).optional().describe("Processing options"),
  }),
  execute: async (args, ctx) => {
    // Validate file access permissions
    await validateFilePath(args.filePath, ctx.sessionID)

    // Read file content
    const content = await Bun.file(args.filePath).text()

    // Process based on operation type
    let result
    switch (args.operation) {
      case "analyze":
        result = await analyzeFile(content, args.options)
        break
      case "transform":
        result = await transformFile(content, args.options)
        break
      case "validate":
        result = await validateFile(content, args.options)
        break
      default:
        throw new Error(`Unknown operation: ${args.operation}`)
    }

    return {
      title: `File ${args.operation}: ${args.filePath}`,
      metadata: {
        fileSize: content.length,
        operation: args.operation,
      },
      output: result,
    }
  },
  formatValidationError: (error) => {
    return `Invalid file processor input: ${error.issues.map(i => i.message).join(', ')}`
  }
  }
})

/**
 * Analyzes file content and returns insights
 */
async function analyzeFile(content: string, options: any): Promise<string> {
  const lines = content.split('\n').length
  const words = content.split(/\s+/).length
  const chars = content.length

  return `Analysis: ${lines} lines, ${words} words, ${chars} characters`
}

/**
 * Transforms file content based on provided options
 */
async function transformFile(content: string, options: any): Promise<string> {
  if (options.uppercase) {
    return content.toUpperCase()
  }
  if (options.lowercase) {
    return content.toLowerCase()
  }
  if (options.json) {
    return JSON.stringify(content, null, 2)
  }
  return content
}

/**
 * Validates file against configurable rules
 */
async function validateFile(content: string, options: any): Promise<string> {
  const issues = []

  if (options.maxLength && content.length > options.maxLength) {
    issues.push(`File exceeds maximum length of ${options.maxLength}`)
  }

  if (options.requiredPattern && !new RegExp(options.requiredPattern).test(content)) {
    issues.push(`File does not match required pattern: ${options.requiredPattern}`)
  }

  if (issues.length > 0) {
    throw new Error(`Validation failed: ${issues.join(', ')}`)
  }

  return "Validation passed"
}
```

### Plugin Registration

```typescript
// packages/opencode/src/plugin/registry.ts
export class PluginRegistry {
  private static plugins = new Map<string, Plugin>()

  static register(plugin: Plugin) {
    this.plugins.set(plugin.name, plugin)

    // Initialize plugin
    if (plugin.onInit) {
      plugin.onInit().catch((error) => {
        console.error(`Failed to initialize plugin ${plugin.name}:`, error)
      })
    }
  }

  static async triggerHook(hookName: string, ...args: any[]) {
    const promises = []

    for (const plugin of this.plugins.values()) {
      const hook = plugin[hookName as keyof Plugin]
      if (typeof hook === "function") {
        promises.push((hook as Function).call(plugin, ...args))
      }
    }

    return Promise.allSettled(promises)
  }

  static getPlugin(name: string): Plugin | undefined {
    return this.plugins.get(name)
  }

  static listPlugins(): Plugin[] {
    return Array.from(this.plugins.values())
  }
}
```

```typescript
// packages/opencode/src/provider/custom.ts
export class CustomProvider implements Provider.Interface {
  name = "custom"
  models = [
    {
      id: "custom-model-1",
      name: "Custom Model 1",
      provider: "custom",
      cost: {
        input: 0.001,
        output: 0.002,
        cache_read: 0.0001,
        cache_write: 0.0002,
      },
      context: 128000,
      limit: {
        output: 4096,
      },
    },
  ]

  constructor(private config: Provider.Config) {}

  async getModel(modelID: string): Promise<Provider.Model> {
    const model = this.models.find((m) => m.id === modelID)
    if (!model) {
      throw new Provider.ModelNotFoundError(modelID)
    }
    return model
  }

  createLanguageModel(model: Provider.Model): LanguageModel {
    return createCustomLanguageModel(model, this.config)
  }

  async authenticate(): Promise<boolean> {
    // Custom authentication logic
    if (!this.config.apiKey) {
      return false
    }

    try {
      const response = await fetch(`${this.config.baseURL}/auth/verify`, {
        headers: {
          Authorization: `Bearer ${this.config.apiKey}`,
        },
      })

      return response.ok
    } catch {
      return false
    }
  }
}

function createCustomLanguageModel(model: Provider.Model, config: Provider.Config): LanguageModel {
  return {
    specification: "v1",
    provider: "custom",
    modelId: model.id,
    defaultObjectGenerationMode: "json",

    async doGenerate(options) {
      const response = await fetch(`${config.baseURL}/v1/completions`, {
        method: "POST",
        headers: {
          Authorization: `Bearer ${config.apiKey}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          model: model.id,
          messages: options.prompt,
          max_tokens: options.maxTokens,
          temperature: options.temperature,
        }),
      })

      if (!response.ok) {
        throw new Error(`Custom provider error: ${response.statusText}`)
      }

      const result = await response.json()

      return {
        text: result.choices[0].message.content,
        usage: {
          promptTokens: result.usage.prompt_tokens,
          completionTokens: result.usage.completion_tokens,
        },
        finishReason: result.choices[0].finish_reason,
      }
    },
  }
}
```

```typescript
export class WebhookIntegration {
  constructor(private config: WebhookConfig) {}

      },
    })
  }

    await this.sendEvent({
      type: "tool.executed",
      data: {
        tool,
        result: result.success ? "success" : "error",
        duration: result.duration,
      },
    })
  }
```

## Best Practices

### Tool Development

1. **Parameter Validation**: Use Zod schemas for strict validation
2. **Error Handling**: Provide clear error messages
3. **Progress Updates**: Use metadata for real-time feedback
4. **Permission Checks**: Always validate permissions before execution
5. **Resource Cleanup**: Clean up resources in finally blocks

### Agent Development

1. **Clear Purpose**: Define specific use cases
2. **Appropriate Permissions**: Limit access to necessary tools
3. **Consistent Behavior**: Ensure predictable responses
4. **Error Recovery**: Handle failures gracefully
5. **Documentation**: Provide clear descriptions and examples

### Plugin Development

1. **Lifecycle Management**: Properly initialize and cleanup
2. **Error Isolation**: Don't let plugin errors crash system
3. **Performance**: Minimize overhead and resource usage
4. **Configuration**: Make plugins configurable
5. **Compatibility**: Ensure version compatibility

### Integration Development

1. **API Design**: Use RESTful principles
2. **Authentication**: Secure API endpoints
3. **Rate Limiting**: Prevent abuse
4. **Monitoring**: Log integration events
5. **Documentation**: Provide clear API documentation

## Publishing Extensions

### Extension Registry

```json
{
  "name": "my-custom-tool",
  "version": "1.0.0",
  "description": "A custom tool for OpenCode",
  "author": "Your Name",
  "license": "MIT",
  "repository": "https://github.com/username/my-custom-tool",
  "keywords": ["opencode", "tool", "ai"],
  "type": "tool",
  "main": "dist/index.js",
  "files": ["dist/**/*"],
  "opencode": {
    "minVersion": "1.0.0",
    "maxVersion": "2.0.0",
    "permissions": ["file-read", "network"],
    "config": {
      "apiKey": {
        "type": "string",
        "required": true,
        "description": "API key for external service"
      }
    }
  }
}
```

### Installation

```bash
# Install from registry
opencode extensions install my-custom-tool

# Install from GitHub
opencode extensions install github:username/my-custom-tool

# Install from local file
opencode extensions install ./my-custom-tool.tgz

# List installed extensions
opencode extensions list

# Remove extension
opencode extensions uninstall my-custom-tool
```

The extension system provides a powerful way to customize and enhance OpenCode functionality while maintaining compatibility and security.
