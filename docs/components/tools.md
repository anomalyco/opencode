# Tools Component

The Tools component provides extensible system for AI agent capabilities, including file operations, shell execution, web access, and code intelligence.

## Tool Interface

### Tool Definition

```typescript
/**
 * Defines the contract for all OpenCode tools
 * Provides type safety, validation, and execution context management
 *
 * @template Parameters - Zod schema type for tool input parameters
 * @template Metadata - Type for tool-specific execution metadata
 */
export interface Info<Parameters extends z.ZodType = z.ZodType, M extends Metadata = Metadata> {
  /** Unique identifier for the tool (used in tool calls) */
  id: string

  /** Async initialization function that returns tool configuration */
  init: () => Promise<{
    /** Human-readable description of what the tool does */
    description: string

    /** Zod schema for validating tool input parameters */
    parameters: Parameters

    /** Core execution function with context and error handling */
    execute(
      args: z.infer<Parameters>,
      ctx: Context,
    ): Promise<{
      /** Human-readable title for the tool execution */
      title: string

      /** Tool-specific metadata for tracking and analytics */
      metadata: M

      /** Primary text output from the tool */
      output: string

      /** Optional file attachments generated by the tool */
      attachments?: MessageV2.FilePart[]
    }>

    /** Optional custom error formatter for validation failures */
    formatValidationError?(error: z.ZodError): string
  }>
}
```

### Execution Context

```typescript
/**
 * Provides execution context and utilities for tool implementations
 * Enables real-time updates, cancellation, and metadata tracking
 *
 * @template Metadata - Type for tool-specific execution metadata
 */
export type Context<M extends Metadata = Metadata> = {
  /** Session identifier for the current conversation */
  sessionID: string

  /** Message identifier for the current AI response */
  messageID: string

  /** Name of the agent executing the tool */
  agent: string

  /** AbortSignal for graceful cancellation of long-running operations */
  abort: AbortSignal

  /** Unique identifier for this specific tool call instance */
  callID?: string

  /** Additional context data (model info, etc.) */
  extra?: { [key: string]: any }

  /** Function to update tool execution metadata in real-time */
  metadata(input: { title?: string; metadata?: M }): void
}
```

### Tool Creation Helper

```typescript
/**
 * Helper function to create well-defined tools with automatic validation
 * Wraps user-provided logic with parameter validation and error handling
 *
 * @param id - Unique tool identifier
 * @param init - Tool configuration or async factory function
 * @returns Complete tool definition ready for registration
 */
export function define<Parameters extends z.ZodType, Result extends Metadata>(
  id: string,
  init: Info<Parameters, Result>["init"] | Awaited<ReturnType<Info<Parameters, Result>["init"]>>,
): Info<Parameters, Result> {
  return {
    id,
    init: async () => {
      // Support both sync and async initialization
      const toolInfo = init instanceof Function ? await init() : init
      const execute = toolInfo.execute

      // Wrap execution with automatic parameter validation
      toolInfo.execute = (args, ctx) => {
        try {
          // Validate all parameters against Zod schema
          toolInfo.parameters.parse(args)
        } catch (error) {
          // Use custom error formatter if provided, otherwise use default
          if (error instanceof z.ZodError && toolInfo.formatValidationError) {
            throw new Error(toolInfo.formatValidationError(error), { cause: error })
          }

          // Provide helpful error message with schema requirements
          throw new Error(
            `The ${id} tool was called with invalid arguments: ${error}.\n` +
              `Please rewrite the input so it satisfies the expected schema.`,
            { cause: error },
          )
        }

        // Execute the original tool logic with validated parameters
        return execute(args, ctx)
      }
      return toolInfo
    },
  }
}
```

### 1. File Operations

#### Read Tool (`packages/opencode/src/tool/read.ts`)

```typescript
/**
 * File reading tool with support for partial content and line ranges
 * Provides secure file access within the current working directory context
 */
export const ReadTool = define("read", {
  description: "Reads a file from the local filesystem with optional line range support",
  parameters: z.object({
    filePath: z.string().describe("Absolute path to the file to read").example("/home/user/project/src/main.ts"),
    offset: z
      .number()
      .optional()
      .describe("Line number to start reading from (0-based, for partial reads)")
      .example(10), // Start reading from line 10
    limit: z.number().optional().describe("Maximum number of lines to read (useful for large files)").example(100), // Read first 100 lines
  }),
  execute: async (args, ctx) => {
    // Validate file path is within allowed directory
    await validateFilePath(args.filePath, ctx.sessionID)

    try {
      // Read file content using Bun's optimized file API
      const content = await Bun.file(args.filePath).text()

      // Calculate file statistics for metadata
      const stats = await Bun.file(args.filePath).stat()

      return {
        title: `Read ${args.filePath}`,
        metadata: {
          size: stats.size, // File size in bytes
          lines: content.split("\n").length, // Total line count
          lastModified: stats.lastModified, // Last modification timestamp
        },
        output: content,
      }
    } catch (error) {
      // Provide context-specific error messages
      if (error.code === "ENOENT") {
        throw new Error(`File not found: ${args.filePath}`)
      }
      if (error.code === "EACCES") {
        throw new Error(`Permission denied: ${args.filePath}`)
      }
      throw error
    }
  },
})
```

```typescript
export const BashTool = define("bash", {
  description: "Executes bash commands in the current directory",
  parameters: z.object({
    command: z.string().describe("The bash command to execute"),
  }),
  execute: async (args, ctx) => {
    // Permission check
    await Permission.check({
      tool: "bash",
      command: args.command,
      agent: ctx.agent,
      sessionID: ctx.sessionID,
    })

    // Execute command
    const proc = Bun.spawn({
      cmd: ["bash", "-c", args.command],
      cwd: Instance.directory,
      stdout: "pipe",
      stderr: "pipe",
    })

    const output = await new Response(proc.stdout).text()
    const error = await new Response(proc.stderr).text()

    return {
      title: `bash: ${args.command}`,
      metadata: { exitCode: proc.exitCode },
      output: output || error,
    }
  },
})
```

```typescript
// packages/opencode/src/mcp/index.ts
export async function tools(): Promise<Record<string, any>> {
  const mcpTools = {}

  for (const [name, server] of Object.entries(servers)) {
    if (!server.tools) continue

    for (const [toolName, tool] of Object.entries(server.tools)) {
      const key = `${name}.${toolName}`
      mcpTools[key] = {
        ...tool,
        execute: async (args, opts) => {
          // MCP tool execution
          return await server.callTool(toolName, args)
        },
      }
    }
  }

  return mcpTools
}
```

## Permission Integration

### Permission Checking

```typescript
async function checkPermission(tool: string, args: any, agent: string): Promise<void> {
  const agentInfo = await Agent.get(agent)
  const permission = agentInfo.permission[tool]

  if (permission === "deny") {
    throw new Error(`Tool ${tool} is not allowed for agent ${agent}`)
  }

  if (permission === "ask") {
    // Request user permission
    const response = await Permission.request({
      tool,
      args,
      agent,
      title: `Allow ${agent} to use ${tool}?`,
    })

    if (response === "reject") {
      throw new Error(`User denied permission for ${tool}`)
    }
  }

  // For bash commands, check specific command permissions
  if (tool === "bash" && typeof agentInfo.permission.bash === "object") {
    const cmd = args.command.split(" ")[0]
    const cmdPermission = agentInfo.permission.bash[cmd] || agentInfo.permission.bash["*"]

    if (cmdPermission === "deny") {
      throw new Error(`Command ${cmd} is not allowed for agent ${agent}`)
    }

    if (cmdPermission === "ask") {
      const response = await Permission.request({
        tool: `bash:${cmd}`,
        args,
        agent,
        title: `Allow execution of command: ${cmd}?`,
      })

      if (response === "reject") {
        throw new Error(`User denied execution of ${cmd}`)
      }
    }
  }
}
```

## Tool Execution Flow

### Execution Pipeline

```typescript
// In session/prompt.ts
tools[item.id] = tool({
  id: item.id,
  description: item.description,
  inputSchema: jsonSchema(schema),
  async execute(args, options) {
    // 1. Pre-execution hook
    await Plugin.trigger(
      "tool.execute.before",
      {
        tool: item.id,
        sessionID: input.sessionID,
        callID: options.toolCallId,
      },
      { args },
    )

    // 2. Permission check
    await checkPermission(item.id, args, input.agent.name)

    // 3. Execute tool
    const result = await item.execute(args, {
      sessionID: input.sessionID,
      abort: options.abortSignal,
      messageID: input.processor.message.id,
      callID: options.toolCallId,
      agent: input.agent.name,
      metadata: async (val) => {
        // 4. Real-time updates
        await Session.updatePart({
          ...part,
          state: { ...part.state, ...val },
        })
      },
    })

    // 5. Post-execution hook
    await Plugin.trigger(
      "tool.execute.after",
      {
        tool: item.id,
        sessionID: input.sessionID,
        callID: options.toolCallId,
      },
      result,
    )

    return result
  },
})
```

```typescript
const resultCache = new Map<string, any>()
export async function cachedExecute(tool: string, args: any) {
  const cacheKey = `${tool}:${JSON.stringify(args)}`
  if (resultCache.has(cacheKey)) {
    return resultCache.get(cacheKey)
  }

  const result = await execute(tool, args)
  resultCache.set(cacheKey, result)
  return result
```
