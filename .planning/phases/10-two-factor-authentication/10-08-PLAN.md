Add endpoint to check if device trust is active:

```typescript
.get("/device-trust/status", async (c) => {
  const authConfig = ServerAuth.get()

  // Check if 2FA is enabled
  if (!authConfig.twoFactorEnabled) {
    return c.json({ twoFactorEnabled: false, deviceTrusted: false })
  }

  // Check for device trust cookie
  const deviceTrustCookie = getCookie(c, "opencode_device_trust")
  if (!deviceTrustCookie) {
    return c.json({ twoFactorEnabled: true, deviceTrusted: false })
  }

  // Verify the cookie is valid (without checking user - just structure)
  const fingerprint = createDeviceFingerprint(c.req.header("User-Agent") ?? "")
  const trustedUser = await verifyDeviceTrustToken(
    deviceTrustCookie,
    fingerprint,
    getTokenSecret(),
  )

  return c.json({
    twoFactorEnabled: true,
    deviceTrusted: trustedUser !== null,
  })
})
```
