// Browser driver — Fase 4 (browser agent nativo).
//
// Runs under Node.js because Playwright does not support the bun runtime
// (verified: its launcher and WebSocket client hang under bun, while plain
// Node.js launches Chromium in well under a second). The browser tool writes
// this source to the temp dir and spawns:
//
//   node driver.mjs <playwright-entry-url> <scratch-dir>
//
// Protocol — JSON lines on stdin/stdout:
//   in : { id, action, params }
//   out: { id, ok: true, result } | { id, ok: false, error }
//
// stdout carries ONLY protocol messages; diagnostics go to stderr. The driver
// exits after answering "close", and cleans up the browser when stdin closes
// (tool/process died) or on SIGTERM/SIGINT.
import fs from "node:fs"
import path from "node:path"
import readline from "node:readline"

const entry = process.argv[2]
const scratch = process.argv[3]

const MAX_CONSOLE = 200
const MAX_NETWORK = 500
const MAX_DOWNLOADS = 100

const state = {
  browser: null,
  page: null,
  console: [],
  network: [],
  downloads: [],
}

const cap = (entries, entry, max) => {
  entries.push(entry)
  if (entries.length > max) entries.splice(0, entries.length - max)
}

const stamp = (at) => new Date(at).toISOString().slice(11, 23)

const requirePage = (action) => {
  if (state.page === null)
    throw new Error(`The browser action "${action}" requires an open browser session. Call action "open" first.`)
  return state.page
}

const navigate = async (url, timeout) => {
  await state.page.goto(url, { timeout })
  const title = await state.page.title()
  return `Navigated to ${state.page.url()}\nTitle: ${title}`
}

// Subscribes the page events that back evidence gathering: console output,
// uncaught page errors, network responses/failures, and downloads (copied to
// the scratch dir).
const observePage = () => {
  state.page.on("console", (message) => {
    const location = message.location()
    cap(
      state.console,
      {
        at: Date.now(),
        type: message.type(),
        text: message.text(),
        ...(location?.url ? { url: location.url, line: location.lineNumber } : {}),
      },
      MAX_CONSOLE,
    )
  })
  state.page.on("pageerror", (error) =>
    cap(state.console, { at: Date.now(), type: "pageerror", text: error.message }, MAX_CONSOLE),
  )
  state.page.on("response", (response) => {
    const request = response.request()
    cap(
      state.network,
      {
        at: Date.now(),
        method: request.method(),
        url: response.url(),
        resourceType: request.resourceType(),
        ok: response.ok(),
        status: response.status(),
      },
      MAX_NETWORK,
    )
  })
  state.page.on("requestfailed", (request) =>
    cap(
      state.network,
      {
        at: Date.now(),
        method: request.method(),
        url: request.url(),
        resourceType: request.resourceType(),
        ok: false,
        failure: request.failure()?.errorText ?? "request failed",
      },
      MAX_NETWORK,
    ),
  )
  state.page.on("download", (download) => {
    const filename = path.basename(download.suggestedFilename())
    const target = path.join(scratch, filename)
    const failed = (error) =>
      cap(state.downloads, { at: Date.now(), filename, error: String(error) }, MAX_DOWNLOADS)
    download
      .path()
      .then(
        (source) =>
          fs.promises
            .mkdir(scratch, { recursive: true })
            .then(() => fs.promises.copyFile(source, target))
            .then(
              () => cap(state.downloads, { at: Date.now(), filename, path: target }),
              failed,
            ),
        failed,
      )
  })
}

const history = async (action, direction, timeout) => {
  const page = requirePage(action)
  const response = direction === "back" ? await page.goBack({ timeout }) : await page.goForward({ timeout })
  return {
    title: `browser ${action}`,
    output:
      response === null
        ? `No history entry to go ${action}. Still at ${page.url()}`
        : `Went ${action}. Now at ${page.url()}`,
  }
}

const handlers = {
  open: async (params) => {
    const timeout = params.timeoutMs ?? 30000
    if (state.browser !== null) {
      const navigation = params.url ? await navigate(params.url, timeout) : ""
      return {
        title: "browser open",
        output: [`Browser already open at ${state.page.url()}.`, navigation].filter(Boolean).join("\n"),
      }
    }
    const { chromium } = await import(entry)
    const headless = params.headless ?? true
    const browser = await chromium.launch({ headless })
    try {
      state.page = await browser.newPage()
    } catch (error) {
      // A page that never opened would leak the browser process.
      await browser.close().catch(() => undefined)
      throw error
    }
    state.browser = browser
    observePage()
    const navigation = params.url ? await navigate(params.url, timeout) : ""
    return {
      title: "browser open",
      output: [
        `Browser opened (headless=${headless}).`,
        navigation || `Call action "goto" with a url to navigate. Current page: ${state.page.url()}`,
      ].join("\n"),
    }
  },

  close: async () => {
    if (state.browser === null) return { title: "browser close", output: "No browser session to close." }
    await state.browser.close()
    state.browser = null
    state.page = null
    state.console = []
    state.network = []
    state.downloads = []
    return { title: "browser close", output: "Browser closed. Console, network, and download buffers cleared." }
  },

  goto: async (params) => {
    requirePage("goto")
    const summary = await navigate(params.url, params.timeoutMs ?? 30000)
    return { title: `browser goto ${params.url}`, output: summary }
  },

  back: (params) => history("back", "back", params.timeoutMs ?? 30000),
  forward: (params) => history("forward", "forward", params.timeoutMs ?? 30000),

  click: async (params) => {
    const page = requirePage("click")
    await page.click(params.selector, { timeout: params.timeoutMs ?? 30000 })
    return {
      title: `browser click ${params.selector}`,
      output: `Clicked "${params.selector}". Page: ${page.url()}`,
    }
  },

  fill: async (params) => {
    const page = requirePage("fill")
    await page.fill(params.selector, params.value, { timeout: params.timeoutMs ?? 30000 })
    return { title: `browser fill ${params.selector}`, output: `Filled "${params.selector}".` }
  },

  select: async (params) => {
    const page = requirePage("select")
    await page.selectOption(params.selector, params.value, { timeout: params.timeoutMs ?? 30000 })
    return {
      title: `browser select ${params.selector}`,
      output: `Selected "${params.value}" in "${params.selector}".`,
    }
  },

  upload: async (params) => {
    const page = requirePage("upload")
    await page.setInputFiles(params.selector, params.value, { timeout: params.timeoutMs ?? 30000 })
    return { title: `browser upload ${params.selector}`, output: `Attached ${params.value} to "${params.selector}".` }
  },

  screenshot: async (params) => {
    const page = requirePage("screenshot")
    const options = {
      ...(params.path !== undefined ? { path: params.path } : {}),
      ...(params.fullPage === true ? { fullPage: true } : {}),
    }
    const image = await page.screenshot(options)
    if (params.path !== undefined)
      return { title: "browser screenshot", output: `Screenshot saved to ${params.path}` }
    return {
      title: "browser screenshot",
      output: `Screenshot captured (${params.fullPage === true ? "full page" : "viewport"}).`,
      attachments: [
        {
          type: "file",
          mime: "image/png",
          url: `data:image/png;base64,${image.toString("base64")}`,
        },
      ],
    }
  },

  snapshot: async () => {
    const page = requirePage("snapshot")
    const html = await page.content()
    const title = await page.title()
    return {
      title: "browser snapshot",
      output: `URL: ${page.url()}\nTitle: ${title}\n\n${html}`,
    }
  },

  eval: async (params) => {
    const page = requirePage("eval")
    const value = await page.evaluate(params.code)
    const output = typeof value === "string" ? value : (JSON.stringify(value) ?? String(value))
    return { title: "browser eval", output }
  },

  console: () => {
    requirePage("console")
    const entries = state.console.splice(0)
    if (entries.length === 0)
      return { title: "browser console", output: "No console messages since the last read." }
    return {
      title: `browser console (${entries.length})`,
      output: entries
        .map((entry) => {
          const where = entry.url ? ` (${entry.url}:${entry.line ?? 0})` : ""
          return `[${stamp(entry.at)}] ${entry.type}: ${entry.text}${where}`
        })
        .join("\n"),
    }
  },

  network: () => {
    requirePage("network")
    const entries = state.network.splice(0)
    if (entries.length === 0) return { title: "browser network", output: "No network activity since the last read." }
    return {
      title: `browser network (${entries.length})`,
      output: entries
        .map((entry) => {
          const outcome = entry.ok ? `→ ${entry.status}` : `→ ${entry.failure ?? `HTTP ${entry.status ?? "failed"}`}`
          return `[${stamp(entry.at)}] ${entry.method} ${entry.url} ${outcome} (${entry.resourceType})`
        })
        .join("\n"),
    }
  },

  downloads: () => {
    requirePage("downloads")
    const entries = state.downloads.splice(0)
    if (entries.length === 0)
      return { title: "browser downloads", output: "No completed downloads since the last read." }
    return {
      title: `browser downloads (${entries.length})`,
      output: entries
        .map((entry) =>
          entry.path !== undefined
            ? `[${stamp(entry.at)}] ${entry.filename} → ${entry.path}`
            : `[${stamp(entry.at)}] ${entry.filename} FAILED: ${entry.error ?? "unknown error"}`,
        )
        .join("\n"),
    }
  },

  viewport: async (params) => {
    const page = requirePage("viewport")
    if (params.width !== undefined || params.height !== undefined) {
      await page.setViewportSize({ width: params.width, height: params.height })
      return { title: "browser viewport", output: `Viewport set to ${params.width}x${params.height}.` }
    }
    const size = page.viewportSize()
    return {
      title: "browser viewport",
      output: `Viewport: ${size === null ? "unknown" : `${size.width}x${size.height}`}`,
    }
  },

  wait: async (params) => {
    const page = requirePage("wait")
    if (params.selector !== undefined) {
      await page.waitForSelector(params.selector, { timeout: params.timeoutMs ?? 30000 })
      return { title: "browser wait", output: `Selector "${params.selector}" is present.` }
    }
    const ms = Math.min(params.timeoutMs ?? 1000, 30000)
    await new Promise((resolve) => setTimeout(resolve, ms))
    return { title: "browser wait", output: `Waited ${ms}ms.` }
  },
}

// Resolves only once the reply reached the OS pipe: the close path exits the
// process right after answering, and a lost reply would wedge the tool.
const send = (message) =>
  new Promise((resolve) => {
    try {
      process.stdout.write(JSON.stringify(message) + "\n", () => resolve())
    } catch {
      // stdout gone (tool died) — the stdin 'end' handler performs cleanup.
      resolve()
    }
  })

let closing = false
const shutdown = async () => {
  if (closing) return
  closing = true
  if (state.browser !== null) {
    try {
      await state.browser.close()
    } catch {
      // Best effort: the parent is going away regardless.
    }
    state.browser = null
    state.page = null
  }
}

const handle = async (line) => {
  let message
  try {
    message = JSON.parse(line)
  } catch {
    return
  }
  const id = message?.id
  const action = message?.action
  if (typeof id !== "number" || typeof action !== "string") return
  try {
    const handler = handlers[action]
    if (handler === undefined) throw new Error(`Unknown browser action "${action}".`)
    const result = await handler(message.params ?? {})
    await send({ id, ok: true, result })
    if (action === "close") {
      await shutdown()
      process.exit(0)
    }
  } catch (error) {
    await send({ id, ok: false, error: error instanceof Error ? error.message : String(error) })
  }
}

const input = readline.createInterface({ input: process.stdin })
input.on("line", (line) => {
  if (line.trim() !== "") void handle(line)
})
input.on("end", () => {
  // The tool closed our stdin (session discarded or process died): never
  // leave an orphaned browser behind.
  void shutdown().finally(() => process.exit(0))
})
for (const signal of ["SIGTERM", "SIGINT"]) {
  process.on(signal, () => {
    void shutdown().finally(() => process.exit(0))
  })
}
process.on("uncaughtException", (error) => {
  console.error(`browser driver uncaught: ${error?.stack ?? error}`)
  void shutdown().finally(() => process.exit(1))
})
